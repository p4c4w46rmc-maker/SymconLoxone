// Sprint 17: Control-Definition aus aktueller LoxAPP3 laden.
// Diese Methode in LoxoneGateway/module.php innerhalb der Klasse einfügen,
// z. B. nach ReadMiniserver(). Danach ist sie als LOX_GetControlDefinition($id, $uuid) verfügbar.

public function GetControlDefinition(string $controlUuid): array
{
    $controlUuid = strtolower(trim($controlUuid));
    if ($controlUuid === '') {
        throw new RuntimeException('ControlUUID ist leer.');
    }

    $data = $this->CreateApi()->getLoxApp3();
    $controls = is_array($data['controls'] ?? null) ? $data['controls'] : [];

    foreach ($controls as $uuid => $control) {
        if (strtolower((string)$uuid) === $controlUuid) {
            return is_array($control) ? $control : [];
        }
        if (is_array($control) && strtolower((string)($control['uuidAction'] ?? '')) === $controlUuid) {
            return $control;
        }
    }

    throw new RuntimeException('Control nicht in LoxAPP3 gefunden: ' . $controlUuid);
}
